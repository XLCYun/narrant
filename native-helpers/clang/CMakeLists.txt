cmake_minimum_required(VERSION 3.24)
# 仅构建 Narrant helper，LLVM_ROOT 必须是隔离且已核验的固定 SDK。
project(narrant_clang LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LLVM 21.1.8 EXACT REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
# 固定 Windows 生成器已经明确选定 VS2022；修复上游导出的 VS2019 DIA 路径。
if(WIN32)
  include(cmake/RelocateWindowsDia.cmake)
  narrant_relocate_windows_dia("${CMAKE_GENERATOR_INSTANCE}")
endif()
add_executable(narrant-clang src/main.cpp)
target_include_directories(narrant-clang SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
separate_arguments(NARRANT_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(narrant-clang PRIVATE ${NARRANT_LLVM_DEFINITIONS})
if(MSVC)
  target_compile_options(narrant-clang PRIVATE /EHsc /GR- /utf-8)
  target_compile_definitions(narrant-clang PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(narrant-clang PRIVATE -fexceptions -fno-rtti)
endif()
# 优先发布包的整体 Clang 动态库；没有该目标的 Windows SDK 使用官方静态依赖图。
if(TARGET clang-cpp)
  target_link_libraries(narrant-clang PRIVATE clang-cpp)
else()
  target_link_libraries(narrant-clang PRIVATE clangTooling clangDriver clangFrontend clangIndex clangAST clangLex clangBasic)
endif()
# 所有生成器都产生 build/narrant-clang[.exe]，避免 CI 猜测 Debug/Release 子目录。
set_target_properties(narrant-clang PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(NARRANT_CONFIGURATION DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(narrant-clang PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${NARRANT_CONFIGURATION} "${CMAKE_BINARY_DIR}")
endforeach()
if(APPLE)
  set_target_properties(narrant-clang PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/runtime/lib;@executable_path/../runtime/lib")
elseif(UNIX)
  set_target_properties(narrant-clang PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/runtime/lib;$ORIGIN/../runtime/lib")
endif()
install(TARGETS narrant-clang RUNTIME DESTINATION bin)
# Windows 动态依赖放在程序旁边；静态 SDK 不引入多余 DLL。
if(WIN32 AND TARGET clang-cpp)
  get_target_property(NARRANT_CLANG_LIBRARY_TYPE clang-cpp TYPE)
  if(NARRANT_CLANG_LIBRARY_TYPE STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET narrant-clang POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:narrant-clang> $<TARGET_FILE_DIR:narrant-clang>
      COMMAND_EXPAND_LISTS)
  endif()
endif()
